cmake_minimum_required(VERSION 3.16)
project(pcm_ring_tests C)
enable_testing()
find_package(Threads REQUIRED)
add_executable(test_pcm_ring test_pcm_ring.c)
target_include_directories(test_pcm_ring PRIVATE ../../app/src/main/cpp/shared)
target_link_libraries(test_pcm_ring PRIVATE Threads::Threads)
target_compile_options(test_pcm_ring PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME pcm_ring COMMAND test_pcm_ring)
set_tests_properties(pcm_ring PROPERTIES TIMEOUT 60)
