cmake_minimum_required(VERSION 3.16)
project(graphics_rollback_backups C)
enable_testing()
add_executable(test_graphics_rollback_backups main.c)
target_compile_features(test_graphics_rollback_backups PRIVATE c_std_11)
target_compile_definitions(test_graphics_rollback_backups
                           PRIVATE NDEBUG GRAPHICS_CONFIG_TRANSACTION_TESTING)
if(MSVC)
    target_compile_definitions(test_graphics_rollback_backups PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(test_graphics_rollback_backups PRIVATE /W4 /WX)
else()
    target_compile_options(test_graphics_rollback_backups PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME graphics_rollback_backups COMMAND test_graphics_rollback_backups)
